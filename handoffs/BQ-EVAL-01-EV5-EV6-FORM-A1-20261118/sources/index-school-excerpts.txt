        d.assessment.criteria[Number(e.target.dataset.docStandard)].standard=e.target.value; d.standardOverrides={};
        const description=box.querySelector(`[data-doc-area-info="${e.target.dataset.docStandard}"]`);
        if(description) description.textContent=schoolDocumentAreaText(e.target.value,PL()==='de');
      }
      if(e.target.dataset.docCriterion!==undefined){
        const row=d.assessment.criteria[Number(e.target.dataset.docCriterion)];
        row.criterion=e.target.value; row.evidence='';
        e.target.closest('fieldset')?.querySelector('details')?.remove();
      }
      if(e.target.dataset.docRating!==undefined){
        const row=d.assessment.criteria[Number(e.target.dataset.docRating)];
        row.rating=e.target.value; row.evidence=''; d.standardOverrides={};
        e.target.closest('fieldset')?.querySelector('details')?.remove();
      }
      if(name==='source' && !d.extracted) d.source=e.target.value;
    }
    if(!d.preview && ['title','day','standard','rating'].includes(name)){
      if(name==='rating' && !schoolEvaluationNeedsPractice(d.rating)) d.focus=schoolEvaluationNeedsPractice(e.target.value);
      d[name]=e.target.value;
    }
    if(d.preview && e.target.dataset.docOverride!==undefined){
      const code=e.target.dataset.docOverride; d.standardOverrides=d.standardOverrides||{};
      if(['RS','C','M','E'].includes(e.target.value)) d.standardOverrides[code]=e.target.value; else delete d.standardOverrides[code];
      redraw(); body.querySelector(`[data-doc-override="${code}"]`)?.focus();
    }
    if(d.preview && name==='exerciseFocus'){ d.focus=e.target.checked; redraw(); body.querySelector('#pSchoolEvaluation [name="exerciseFocus"]')?.focus(); }
  });
  box.addEventListener('submit',e=>{
    e.preventDefault();
    const d=current(), de=PL()==='de';
    if(!d || d.busy) return;
    if(!d.preview){
      d.title=d.title.trim();
      if(d.assessment){
        if(!d.assessment.criteria.some(c=>Object.hasOwn(c,'standard'))){
          if(d.standard==='unmapped'){ d.rating='N'; d.focus=false; }
          else if(!schoolEvaluationNeedsPractice(d.rating)) d.focus=false;
        }
        const payload=schoolEvaluationPayload(d);
        const errors=checkEvent({id:'e_000000000000000000000001',v:1,type:'school_evaluation',at:new Date().toISOString(),author:'p_Validation',device:'d_Validation',payload});
        if(errors.length || !(SUBJECTS.some(s=>s.key===d.assessment.area) || d.assessment.area==='math' && !d.assessment.criteria.some(c=>Object.hasOwn(c,'standard'))) || d.day>dayKey()){
          d.error=de?'Bitte Titel, Datum, Fach, Kriterien und Lernbereiche prüfen. Fehlende Einstufungen dürfen leer bleiben.':'Check the title, date, subject, criteria and learning areas. Missing ratings may stay empty.';
        } else { d.error=''; d.preview=true; }
      } else if(!d.title || /[\x00-\x1f\x7f]/.test(d.title) || !d.day || d.day>dayKey() || !Object.hasOwn(SCHOOL_EVALUATION_TARGETS,d.standard) || !SL[d.rating]){
        d.error=de?'Bitte prüfen Sie Titel, Datum, Kriterium und Einstufung.':'Check the title, date, criterion and rating.';
      } else { d.error=''; d.preview=true; d.focus=schoolEvaluationNeedsPractice(d.rating) && d.focus; }
      redraw(); body.querySelector('#pSchoolEvaluation button[type="submit"]')?.focus(); return;
    }
    const {child,evaluation,revision,title,day,standard,rating}=d;
    const payload=schoolEvaluationPayload(d);
    if(d.base){
      const active=schoolFromAccount(child)?.tasks.find(t=>t.manual?.evaluation===evaluation)?.manual;
      const conflicts=active?.conflicts||[];
      if(active?.revision!==d.base || conflicts.length!==(d.resolves||[]).length || conflicts.some(id=>!d.resolves.includes(id))){
        d.error=de?'Die Bewertung hat sich beim Abgleich geändert. Brechen Sie den Entwurf ab und öffnen Sie die aktuelle Fassung.':'The assessment changed during sync. Cancel this draft and open the current revision.';
        redraw(); return;
      }
      payload.base=d.base;
      if(d.resolves) payload.resolves=d.resolves;
      if(d.withdrawn){ payload.withdrawn=true; if(!d.assessment || !d.assessment.criteria.some(c=>Object.hasOwn(c,'standard'))) payload.focus='none'; }
    }
    const event=eventHook('school_evaluation',payload,child);
    if(!event || !(evRaw('bq2_'+child+'__events')||[]).some(x=>x.id===event.id)){
      evNeuLaden(); d.error=de?'Speichern fehlgeschlagen. Ihre Eingaben sind noch da.':'Saving failed. Your entries are still here.';
      redraw(); return;
    }
    schoolEvaluationNotice={child,author:d.author,revision};
    schoolEvaluationDraft=null;
    state.sopen=[];
    renderSheets();
    body.querySelector('#pSchoolEvaluation [data-eval-action="open"]')?.focus();
  });
}
