function bindSchoolEvaluation(body){
  const box=body.querySelector('#pSchoolEvaluation');
  if(!box) return;
  const redraw=()=>{
    const activeBox=body.querySelector("#pSchoolEvaluation");
    if(!activeBox) return;
    activeBox.outerHTML=schoolEvaluationHTML();
    bindSchoolEvaluation(body);
  };
  const current=()=>{
    const t=schoolEvaluationTarget(), d=schoolEvaluationDraft;
    return t && d?.child===t.id && d.author===t.author ? d : null;
  };
  box.addEventListener('click',async e=>{
    const remove=e.target.closest('[data-doc-remove]');
    if(remove && current()?.assessment && !current().preview && !current().busy){
      const d=current(), i=Number(remove.dataset.docRemove);
      if(d.assessment.criteria.length>1){ d.assessment.criteria.splice(i,1); if(Array.isArray(d.touched)) d.touched.splice(i,1); }
      d.standardOverrides={}; redraw(); body.querySelector('[data-eval-action="add-criterion"]')?.focus(); return;
    }
    const action=e.target.closest('[data-eval-action]')?.dataset.evalAction, t=schoolEvaluationTarget();
    if(!action || !t) return;
    if(action==='remap'){
      // Same revision draft as an ordinary correction: same evaluation id, base = active revision, preview only.
      if(schoolEvaluationDraft) return;
      const id=e.target.closest('[data-eval-action]').dataset.evalId;
      const tasks=schoolFromAccount(t.id)?.tasks||[], cand=evUebRemapCandidates(tasks).find(c=>c.evaluation===id);
      const m=cand && tasks.find(x=>x.manual?.evaluation===id)?.manual, p=m && m.history.find(x=>x.revision===m.revision);
      if(!p?.assessment) return;
      schoolEvaluationNotice=null;
      schoolEvaluationDraft={child:t.id,author:t.author,evaluation:m.evaluation,revision:crypto.randomUUID(),
        base:m.revision,resolves:null,title:p.title,day:p.day,standard:p.standard,rating:p.rating,focus:false,preview:true,withdrawn:false,
        assessment:evUebApplyCandidate(p.assessment,cand.rows),
        ...(p.standardOverrides?{standardOverrides:{...p.standardOverrides}}:{})};
      redraw(); body.querySelector('#pSchoolEvaluation button[type="submit"]')?.focus(); return;
    }
    if(action==='open'){
      schoolEvaluationNotice=null;
      schoolEvaluationCapture=null; schoolEvaluationDraft=schoolEvaluationNew(t);
    } else if(action==='add-criterion' && current()?.assessment && !current().preview && !current().busy){
      const criteria=current().assessment.criteria, criterion=schoolDocumentNew().criteria[0];
      if(!criteria.some(c=>Object.hasOwn(c,'standard'))) delete criterion.standard;
      if(criteria.length<SCHOOL_EVALUATION_CRITERIA_MAX){ const d=current(); d.touched=criteria.map((_,k)=>evUebTouched(d.touched,k)); criteria.push(criterion); d.touched.push(false); }
    } else if(action==='map-criteria' && current()?.assessment && !current().preview && !current().busy){
      const d=current(); d.assessment.criteria.forEach(c=>c.standard='none');
      if(d.assessment.area==='math') d.assessment.area='maths';
      delete d.standard; delete d.rating; delete d.focus; d.standardOverrides={};
    } else if(action==='ai' && current()?.assessment){
      const d=current(), submitted=d.source;
      if(d.busy || !submitted?.trim()) return;
      d.busy=true; d.error=''; redraw();
      try {
        const bridge=window.webkit?.messageHandlers?.bigQuestEvaluation;
        if(!bridge) throw new Error(PL()==='de'?'KI-Übernahme ist in der iPad-Test-App verfügbar. Die manuelle Eingabe funktioniert hier.':'AI entry is available in the iPad test app. Manual entry works here.');
        const result=await bridge.postMessage({source:submitted});
        if(current()!==d || d.source!==submitted) return;
        if(!result || ['title','date','goal','instructions','feedback'].some(key=>typeof result[key]!=='string') || !Array.isArray(result.criteria) || !result.criteria.length || result.criteria.length>SCHOOL_EVALUATION_CRITERIA_MAX || result.criteria.some(c=>!c || typeof c.criterion!=='string' || !c.criterion.trim() || c.criterion.length>2000 || !['','RS','C','M','E'].includes(c.rating) || typeof c.evidence!=='string' || c.evidence.length>4000)) throw new Error(PL()==='de'?'Ungültiger KI-Entwurf.':'Invalid AI draft.');

        row.criterion=e.target.value; row.evidence='';
        const k=Number(e.target.dataset.docCriterion);
        if(Object.hasOwn(row,'standard') && !evUebTouched(d.touched,k)){

        }
        e.target.closest('fieldset')?.querySelector('details')?.remove();
      }
      if(e.target.dataset.docRating!==undefined){
        const row=d.assessment.criteria[Number(e.target.dataset.docRating)];
        row.rating=e.target.value; row.evidence=''; d.standardOverrides={};
        e.target.closest('fieldset')?.querySelector('details')?.remove();
      }
      if(name==='source' && !d.extracted) d.source=e.target.value;
    }
    if(!d.preview && ['title','day','standard','rating'].includes(name)){
      if(name==='rating' && !schoolEvaluationNeedsPractice(d.rating)) d.focus=schoolEvaluationNeedsPractice(e.target.value);
      d[name]=e.target.value;
    }

      const code=e.target.dataset.docOverride; d.standardOverrides=d.standardOverrides||{};
      if(['RS','C','M','E'].includes(e.target.value)) d.standardOverrides[code]=e.target.value; else delete d.standardOverrides[code];
